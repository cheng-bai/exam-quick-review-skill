% Measure without forcing tall questions into an unbreakable minipage.
% The tall branch unpacks the same vertical list, preserving natural page breaks.
\newbox\transcriptquestionbox
\newdimen\transcriptquestionheight
\newenvironment{transcriptquestion}{%
  \par\setbox\transcriptquestionbox=\vbox\bgroup
  \hsize=\linewidth
}{%
  \par\egroup
  \transcriptquestionheight=\ht\transcriptquestionbox
  \advance\transcriptquestionheight by \dp\transcriptquestionbox
  \ifdim\transcriptquestionheight<.94\textheight
    \Needspace{\transcriptquestionheight}\box\transcriptquestionbox
  \else
    \typeout{Transcript: long question uses natural page breaks.}%
    \unvbox\transcriptquestionbox
  \fi
  \par\medskip
}
